package main

import (
	"fmt"
	"log"

	"github.com/xuri/excelize/v2"
)

func main() {

	f := excelize.NewFile()

	lineWidth := 1.2

	err := f.AddShape("Sheet1",
		&excelize.Shape{
			Cell: "G6",
			Type: "rect",
			Line: excelize.ShapeLine{Color: "4286F4", Width: &lineWidth},
			Fill: excelize.Fill{Color: []string{"8EB9FF"}},
			Paragraph: []excelize.RichTextRun{
				{
					Text: "Rectangle Shape",
					Font: &excelize.Font{
						Bold:      true,
						Italic:    true,
						Family:    "Times New Roman",
						Size:      18,
						Color:     "777777",
						Underline: "sng",
					},
				},
			},
			Width:  180,
			Height: 40,
		},
	)
	_ = err



	// 保存
	if err := f.SaveAs("connector_auto.xlsx"); err != nil {
		log.Fatal(err)
	}
	fmt.Println("connector_auto.xlsx を作成しました")
}



package main

import (
	"fmt"
	"log"

	"github.com/xuri/excelize/v2"
)

// セル座標をピクセル位置に変換する（簡易版）
func cellToPixels(col, row int) (x, y int) {
	// Excel のデフォルト列幅は約 64px、高さは約 20px（フォントによって変動）
	const colWidth = 64
	const rowHeight = 20
	x = (col - 1) * colWidth
	y = (row - 1) * rowHeight
	return
}

func main() {
	f := excelize.NewFile()

	// 開始セルと終了セル
	startCell := "A1"
	endCell := "D5"

	// セル座標を列番号・行番号に変換
	startCol, startRow, err := excelize.CellNameToCoordinates(startCell)
	if err != nil {
		log.Fatal(err)
	}
	endCol, endRow, err := excelize.CellNameToCoordinates(endCell)
	if err != nil {
		log.Fatal(err)
	}

	// ピクセル座標に変換
	startX, startY := cellToPixels(startCol, startRow)
	endX, endY := cellToPixels(endCol, endRow)

	// コネクタ図形 JSON
	connector := fmt.Sprintf(`{
		"type": "connector",
		"connector": {
			"shape": "straight"
		},
		"style": {
			"line": {
				"color": "#FF0000",
				"width": 2
			}
		},
		"from": {
			"x": %d,
			"y": %d
		},
		"to": {
			"x": %d,
			"y": %d
		}
	}`, startX, startY, endX, endY)

	// コネクタ追加
	if err := f.AddShape("Sheet1", "A1", connector); err != nil {
		log.Fatal(err)
	}

	// 保存
	if err := f.SaveAs("connector_auto.xlsx"); err != nil {
		log.Fatal(err)
	}

	fmt.Println("connector_auto.xlsx を作成しました")
}
