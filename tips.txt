package main

package main

import (
	"bufio"
	"fmt"
	"os"
	"strings"
)

type Block struct {
	Name     string
	Content  string
	Children []*Block
}

func main() {
	root := parseFile("input.txt")
	printAST(root, 0)
}

func parseFile(path string) *Block {
	f, err := os.Open(path)
	if err != nil {
		panic(err)
	}
	defer f.Close()

	scanner := bufio.NewScanner(f)

	root := &Block{Name: "root"}
	stack := []*Block{root}

	for scanner.Scan() {
		line := scanner.Text()
		trim := strings.TrimSpace(line)

		// ブロック開始
		if strings.Contains(trim, "{") {
			beforeBrace := strings.TrimSpace(trim[:strings.Index(trim, "{")])

			newBlock := &Block{Name: beforeBrace}
			parent := stack[len(stack)-1]
			parent.Children = append(parent.Children, newBlock)

			stack = append(stack, newBlock)

			// 行末に残りのテキストがあれば Content に入れる
			after := strings.TrimSpace(trim[strings.Index(trim, "{")+1:])
			if after != "" && after != "}" {
				newBlock.Content += after + "\n"
			}

			// 単行で閉じている場合
			if strings.Contains(trim, "}") {
				stack = stack[:len(stack)-1]
			}

			continue
		}

		// ブロック終了
		if strings.Contains(trim, "}") {
			stack = stack[:len(stack)-1]
			continue
		}

		// ブロック内の通常行
		if len(stack) > 1 {
			stack[len(stack)-1].Content += trim + "\n"
		}
	}

	return root
}

func printAST(b *Block, indent int) {
	prefix := strings.Repeat("  ", indent)
	fmt.Printf("%sBlock: %s\n", prefix, b.Name)
	if strings.TrimSpace(b.Content) != "" {
		fmt.Printf("%s  Content:\n%s%s\n", prefix, prefix, indentText(b.Content, prefix+"    "))
	}
	for _, child := range b.Children {
		printAST(child, indent+1)
	}
}

func indentText(text, indent string) string {
	lines := strings.Split(text, "\n")
	for i := range lines {
		lines[i] = indent + lines[i]
	}
	return strings.Join(lines, "\n")
}
